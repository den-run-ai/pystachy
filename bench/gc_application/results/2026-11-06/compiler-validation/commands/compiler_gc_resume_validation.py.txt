def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def run(name, command, extra=None, timeout=900):
    print('START', name, flush=True)
    start = time.monotonic()
    log = LOGS / (name + '.log')
    with log.open('w') as stream:
        process = subprocess.run([str(x) for x in command], cwd=WORK,
                                 env=ENV | (extra or {}), stdout=stream,
                                 stderr=subprocess.STDOUT, timeout=timeout)
    item = {'name': name, 'command': [str(x) for x in command], 'environment': extra or {},
            'elapsed_seconds': time.monotonic() - start, 'exit_code': process.returncode,
            'log': str(log), 'log_sha256': sha(log)}
    report['steps'].append(item)
    (LOGS / 'validation.json').write_text(json.dumps(report, indent=2) + '\n')
    print('DONE', name, 'exit', process.returncode, flush=True)
    print('\n'.join(log.read_text().splitlines()[-5:]), flush=True)
    if process.returncode:
        raise SystemExit(process.returncode)
