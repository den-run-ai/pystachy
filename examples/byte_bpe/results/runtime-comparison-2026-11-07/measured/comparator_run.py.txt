"""Expose the common pattern for the unchanged upstream API, then run a driver.

Usage: python run_mode.py MODE DRIVER [driver arguments...]
All injection and imports occur before the driver's measured regions.
"""
from pathlib import Path
import sys
import runpy

root=Path(__file__).resolve().parent
mode=sys.argv[1]
driver=sys.argv[2]
sys.path.insert(0,str(root/mode))
import bpe_tokeniser
assert bpe_tokeniser.__file__.endswith('.so'), bpe_tokeniser.__file__
bpe_tokeniser.BPE_PATTERN=r"'s|'t|'re|'ve|'m|'ll|'d| ?[a-zA-Z]+| ?\d+| ?[^\sa-zA-Z\d]+|\s+(?!\S)|\s+"
sys.argv=[driver]+sys.argv[3:]
runpy.run_path(driver,run_name='__main__')
