import hashlib
import importlib.util
import json
import os
from pathlib import Path
import subprocess
import time

ROOT = Path(__file__).resolve().parent
REPO = ROOT / 'pystachy'
LLVM = Path('/workspace/scratch/c154e74884e5/llvm-cache/ubuntu-18.1.3-1ubuntu1-amd64/bin')
PYTHON = '/root/.local/share/uv/python/cpython-3.13.16-linux-x86_64-gnu/bin/python3.13'
PYPY = '/root/.local/share/uv/python/pypy-3.11.17-linux-x86_64-gnu/bin/pypy3.11'

def clean_env():
    env = {k: v for k, v in os.environ.items()
           if not k.startswith(('PYSTACHY_', 'PYPY_')) and k not in ('PYTHONPATH', 'PYTHONHOME')}
    env.update(PYTHONHASHSEED='0', PYTHONDONTWRITEBYTECODE='1',
               PYSTACHY_HOME=str(REPO), PYSTACHY_PATH=str(REPO / 'bench/pyperformance'),
               PYSTACHY_LLVM=str(LLVM), PYSTACHY_CACHE=str(ROOT / 'native/runtime-cache'),
               PYSTACHY_PROGRAM_CACHE='0')
    return env

def sha(path):
    return hashlib.sha256(Path(path).read_bytes()).hexdigest()

def main():
    env = clean_env()
    records = []
    jobs = [
        ('runtime-prime', [str(REPO / 'pystachy'), 'run', str(ROOT / 'native/warm.py')], env),
        ('build-stages', [str(REPO / 'pystachy'), 'build', str(ROOT / 'stages_warm.py'), '-o', str(ROOT / 'native/stages')], env),
        ('build-correctness', [str(REPO / 'pystachy'), 'build', str(ROOT / 'correctness.py'), '-o', str(ROOT / 'native/correctness')], env),
        ('correctness-original', [PYTHON, str(ROOT / 'correctness.py')], dict(env, PYTHONPATH=str(ROOT / 'original'))),
        ('correctness-pypy', [PYPY, str(ROOT / 'correctness.py')], dict(env, PYTHONPATH=str(ROOT / 'original'))),
        ('correctness-adapted', [PYTHON, str(ROOT / 'correctness.py')], dict(env, PYTHONPATH=str(REPO / 'bench/pyperformance'))),
        ('correctness-aot', [str(ROOT / 'native/correctness')], env),
        ('correctness-aot-gcstress', [str(ROOT / 'native/correctness')], dict(env, PYSTACHY_GC_STRESS_PROGRAM='1', PYSTACHY_GC_STRESS='1')),
    ]
    for name, cmd, runenv in jobs:
        start = time.perf_counter()
        p = subprocess.run(cmd, env=runenv, cwd=REPO, capture_output=True, timeout=240)
        record = {'name': name, 'command': cmd, 'seconds': time.perf_counter()-start,
                  'returncode': p.returncode, 'stdout_sha256': hashlib.sha256(p.stdout).hexdigest(),
                  'stderr': p.stderr.decode()}
        (ROOT / 'evidence' / (name + '.stdout')).write_bytes(p.stdout)
        (ROOT / 'evidence' / (name + '.stderr')).write_bytes(p.stderr)
        records.append(record)
        (ROOT / 'evidence/preparation.json').write_text(json.dumps(records, indent=2))
        print(name, round(record['seconds'],3), p.returncode, flush=True)
        if p.returncode:
            print(record['stderr'])
            raise SystemExit(p.returncode)
    print('native SHA256', sha(ROOT / 'native/stages'), flush=True)

if __name__ == '__main__':
    main()
